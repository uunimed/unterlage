Supabase SQL ცხრილების ინსტრუქცია

აპლიკაცია ავტომატურად ინახავს ფაილებს როგორც Supabase-ში, ასევე ბრაუზერის IndexedDB საცავში. თუ გსურთ Supabase-ის ცხრილების ხელით ინიციალიზაცია SQL Editor-ში, გამოიყენეთ ეს კოდი: 

CREATE TABLE IF NOT EXISTS public.documents (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  category TEXT NOT NULL,
  file_name TEXT NOT NULL,
  file_type TEXT,
  file_ext TEXT,
  file_size BIGINT,
  file_data_base64 TEXT,
  page_count INT DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.trusted_device (
  id INT PRIMARY KEY DEFAULT 1,
  device_id TEXT,
  device_info TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.auth_requests (
  id TEXT PRIMARY KEY,
  device_id TEXT,
  ip TEXT,
  country TEXT,
  city TEXT,
  device_name TEXT,
  status TEXT DEFAULT 'pending',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.documents DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.trusted_device DISABLE ROW LEVEL SECURITY;
